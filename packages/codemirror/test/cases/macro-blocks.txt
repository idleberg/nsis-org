# If block with ElseIf and Else

Section "Main"
  ${If} $0 != ""
    DetailPrint "yes"
  ${ElseIf} $0 == "x"
    DetailPrint "x"
  ${Else}
    DetailPrint "no"
  ${EndIf}
SectionEnd

# Switch block with Case, Break and Default

  ${Switch} $0
    ${Case} 0
      DetailPrint "ok"
      ${Break}
    ${Default}
      DetailPrint "other"
  ${EndSwitch}
FunctionEnd

# Loop blocks

Function .onInit
  ${Do}
    IntOp $0 $0 - 1
  ${LoopUntil} $0 <= 0
  ${For} $1 1 3
    IntOp $0 $0 + $1
  ${Next}
  ${While} $0 < 3
    IntOp $0 $0 + 1
  ${EndWhile}
  ${Unless} $0 == 1
    Nop
  ${EndUnless}
FunctionEnd
